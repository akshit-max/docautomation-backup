+// POST /api/doc/[id]/preview ΓÇö renders HTML from raw content body (used for live preview without saving)
+export async function POST(request: Request, { params }: { params: Promise<{ id: string }> }) {
+  try {
+    const { id } = await params;
+    const body = await request.json();
+    const content = body.content || {};
+    const templateType = body.template_type || 'developer_doc';
+
+    const ALLOWED_TEMPLATE_TYPES = ['invoice', 'receipt_template', 'client_doc', 'compliance', 'timeline', 'developer_doc'];
+    const safeTemplateType = ALLOWED_TEMPLATE_TYPES.includes(templateType) ? templateType : 'developer_doc';
+
+    const templatePath = path.join(
+      process.cwd(),
+      '..',
+      'backend',
+      'templates',
+      safeTemplateType,
+      'layout.html'
+    );
+
+    let html: string;
+    if (fs.existsSync(templatePath)) {
+      const templateSource = fs.readFileSync(templatePath, 'utf-8');
+      // Re-run calculations on the live content to ensure correct display values
+      // (amounts, subtotal, discount, gst, total) regardless of how they were stored.
+      let computedContent = { ...content };
+      if (safeTemplateType === 'invoice' || safeTemplateType === 'client_doc') {
+        computedContent = calculateTotals(computedContent);
+      } else if (safeTemplateType === 'receipt_template') {
+        computedContent = calculateReceiptTotals(computedContent);
+      }
+      html = renderTemplate(templateSource, { ...computedContent, template_type: safeTemplateType });
+    } else {
+      html = `<!DOCTYPE html><html><body><pre>${JSON.stringify(content, null, 2)}</pre></body></html>`;
+    }
+
+    return new NextResponse(html, {
+      status: 200,
+      headers: { 'Content-Type': 'text/html; charset=utf-8' },
+    });
+  } catch (error) {
+    console.error('Error rendering live preview:', error);
+    return new NextResponse('<h1>Preview failed</h1>', {
+      status: 500,
+      headers: { 'Content-Type': 'text/html' },
+    });
+  }
